      note "{f}"

# a comment at column 0
      warn "still inside each"
  record "back in the stage"
stage "two"
  poll every 10s for 15m
    read tool
      capture s = jq .status
    until s == "done"
    fail s in ["failed", "deleted"]
  ask secret pin "PIN?"
  assert (a or b) and not c > -1.5 else "no"
  confirm "go on?"
