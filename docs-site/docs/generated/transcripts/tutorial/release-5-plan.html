          → $call1$current

━━ Stage 2/4 · Choose the version ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  read-only, no gate; valid --from start
  2.1  read ❯ git ls-remote \
                --tags \
                --refs \
                --sort=-version:refname \
                <remote>
        → last
  2.2  read ❯ git rev-list --count <last ← 2.1>..HEAD
        → ahead
  2.3  assert ahead > 0 else "Nothing to release: HEAD is <last ← 2.1>."
  2.4  read ❯ git log --oneline <last ← 2.1>..HEAD
        → commits [list]
  2.5  each commit in commits
    2.5.1  note "<commit ← 2.5>"
  2.6  read ❯ git rev-parse --short HEAD
        → head
  2.7  ask version "Version to release after <last ← 2.1>?"
  2.8  assert version ~ /^v[0-9]+\.[0-9]+\.[0-9]+$/ else "<version ← 2.7> is not a version like v1.2
.3."

━━ Stage 3/4 · Is it released? ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  read-only, no gate; not a --from start: uses `version` (line 35)
  3.1  git.tag_state({remote}, {version}) as tag
    3.1.1  read ❯ git rev-parse --quiet --verify refs/tags/<version ← 2.7>
          ? tag.untagged
    3.1.2  read ❯ git ls-remote --exit-code --tags <remote> refs/tags/<version ← 2.7>
          ? tag.unpushed
  3.2  when tag.untagged
    3.2.1  note "<version ← 2.7> is not tagged yet."
  3.3  when not tag.untagged and tag.unpushed
    3.3.1  note "<version ← 2.7> is tagged here, but not pushed yet."
  3.4  when not tag.untagged and not tag.unpushed
    3.4.1  note "<version ← 2.7> is already tagged and pushed. Nothing to do."
